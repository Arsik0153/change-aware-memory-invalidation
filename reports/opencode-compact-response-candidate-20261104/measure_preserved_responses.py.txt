"""Measure compact v1 bytes on exactly three already-public first-task responses."""
import hashlib
import json
from pathlib import Path

from compact_response import compact_result
from memory_backend import canonical, digest

STAGE = Path(__file__).resolve().parent
ROOT = next(p for p in STAGE.parents if (p / 'src/cami').is_dir())


def main():
    rows, inputs = [], {}
    for arm in ('static', 'cami'):
        evidence = ROOT / f'runs/opencode-development-v3-20261104/trials/r1-{arm}-s1/session/evidence'
        path, backend_path = evidence / 'messages.json', evidence / 'memory-backend-trace.jsonl'
        for source in (path, backend_path):
            inputs[str(source.relative_to(ROOT))] = hashlib.sha256(source.read_bytes()).hexdigest()
        lookup_metrics = iter(json.loads(line) for line in backend_path.read_text().splitlines() if json.loads(line)['event'] == 'memory_lookup')
        ordinal = 0
        for message in json.loads(path.read_text()):
            for tool in message.get('content', []):
                if tool.get('type') != 'tool':
                    continue
                ordinal += 1
                if tool['name'] != 'research_memory_memory_lookup':
                    continue
                full = {'content': tool['state']['content'], 'isError': False}
                metric = next(lookup_metrics)
                assert len(canonical(full)) == metric['response_bytes']
                compact, receipt = compact_result(full)
                old, new = (json.loads(result['content'][0]['text']) for result in (full, compact))
                assert [entry['id'] for entry in new['entries']] == [entry['id'] for entry in old['entries']]
                assert [entry['payload']['text'] for entry in new['entries']] == [entry['payload']['text'] for entry in old['entries']]
                assert new['freshness'] == old['freshness'] and new['selection_budget'] == old['selection_budget']
                full_text, compact_text = len(full['content'][0]['text'].encode()), len(compact['content'][0]['text'].encode())
                rows.append({'arm': arm, 'ordinal': ordinal, 'full_text_utf8_bytes': full_text,
                             'compact_text_utf8_bytes': compact_text, 'text_bytes_removed': full_text - compact_text,
                             'text_byte_reduction_fraction': (full_text - compact_text) / full_text,
                             'full_mcp_result_canonical_bytes': len(canonical(full)),
                             'compact_mcp_result_canonical_bytes': len(canonical(compact)),
                             'extra_audit_object_canonical_bytes': len(canonical(receipt)),
                             'audit_receipt_sha256': receipt['receipt_sha256'],
                             'full_result_sha256': digest(full), 'compact_result_sha256': digest(compact),
                             'notes_unchanged': True, 'entry_ids_unchanged': True, 'budget_unchanged': True})
    assert len(rows) == 3
    report = {'scope': 'Exactly 2 static and 1 CAMI preserved public first-task responses. Actual compact v1 implementation; full note text including headers retained. No token, billing, model quality, end-to-end speed or trace-storage saving claim.',
              'cost_boundary': 'Compact adds projection/hashing plus full-response receipt trace storage. Projection/hashing/serialization are inside lookup timing; trace append remains outside lookup span as before and must be charged by the outer session harness.',
              'inputs': inputs, 'responses': rows,
              'totals': {key: sum(row[key] for row in rows) for key in ['full_text_utf8_bytes', 'compact_text_utf8_bytes', 'text_bytes_removed', 'extra_audit_object_canonical_bytes']}}
    (STAGE / 'preserved-response-byte-measurement.json').write_text(json.dumps(report, indent=2) + '\n')
    print(json.dumps(report, indent=2))


if __name__ == '__main__':
    main()
