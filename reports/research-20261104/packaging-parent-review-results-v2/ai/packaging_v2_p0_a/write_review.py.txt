items = []
for case, fact in zip(cases, facts):
    assert case['id'] == fact['id']
    assert hashlib.sha256((PACKET / case['source_path']).read_bytes()).hexdigest() == case['source_sha256']
    definitions = fact['observed_definitions']
    unique = len(definitions) == 1
    verdict, reasoning = decisions[case['id']]
    actual = definitions[0]['signature'] if unique and case['kind'] == 'signature_equals' else None
    if actual is not None:
        assert (actual == case['predicate']['expected_signature']) == (verdict == 'supported')
    extracted = dict(
        definition_count=len(definitions),
        unconditional=all(d['unconditional'] for d in definitions) if definitions else None,
        target_rebinding=fact['target_rebinding'] or len(definitions) > 1,
        type_parameters_present=any(d['type_parameters_present'] for d in definitions) if definitions else None,
    )
    if case['kind'] == 'signature_equals':
        extracted['actual_signature'] = actual
    if len(definitions) > 1:
        extracted['observed_definitions'] = [dict(line_start=d['line_start'], line_end=d['line_end'], unconditional=d['unconditional'], actual_signature=d['signature']) for d in definitions]
    evidence = [dict(source_path=case['source_path'], available=True, line_start=1, line_end=fact['line_count'])]
    for definition in definitions:
        evidence.append(dict(source_path=case['source_path'], available=True, line_start=definition['line_start'], line_end=definition['line_end']))
    scope = f"Inspected the entire supplied module (lines 1-{fact['line_count']}) for {case['subject']['path']}:{case['subject']['qualname']}. "
    scope += 'Static syntax only; no execution or runtime-binding claim. '
    if len(definitions) > 1:
        scope += 'target_rebinding includes the repeated function bindings; no additional non-function target binding, wildcard import, or evaluated-class global was found.'
    else:
        scope += 'No conditional or duplicate target definition, target rebinding, wildcard import, or evaluated-class target global was found; deferred function bodies do not define module subjects.'
    items.append(dict(id=case['id'], source_sha256=case['source_sha256'], verdict=verdict, extracted_facts=extracted, reasoning=reasoning, evidence=evidence, scope_notes=scope))
