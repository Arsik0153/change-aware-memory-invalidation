"""Independently inventory packet source syntax before opening cases.json.
Only stdlib; never import/execute supplied modules. Binding inventory is
supplemented by whole-module human/AI reading, not a general Python analyzer.
"""
import ast
import hashlib
import json
import pathlib
import sys

PACKET = pathlib.Path('reports/research-20261104/packaging-parents-v2/packets/packaging-derived-5cbe1e44cc7b')
OUT = pathlib.Path('runs/packaging-v2-review-p24-a-20261104')

def dump(node):
    return None if node is None else ast.dump(node, include_attributes=False)

def signature(node):
    args = node.args
    positional = args.posonlyargs + args.args
    defaults = [None] * (len(positional) - len(args.defaults)) + args.defaults
    params = []
    def add(arg, kind, default=None):
        params.append(dict(name=arg.arg, kind=kind, default_ast=dump(default), annotation_ast=dump(arg.annotation)))
    for i, (arg, default) in enumerate(zip(positional, defaults)):
        add(arg, 'positional_only' if i < len(args.posonlyargs) else 'positional_or_keyword', default)
    if args.vararg:
        add(args.vararg, 'var_positional')
    for arg, default in zip(args.kwonlyargs, args.kw_defaults):
        add(arg, 'keyword_only', default)
    if args.kwarg:
        add(args.kwarg, 'var_keyword')
    return dict(parameters=params, return_annotation_ast=dump(node.returns), decorators_ast=[dump(d) for d in node.decorator_list], is_async=isinstance(node, ast.AsyncFunctionDef))

class Inventory(ast.NodeVisitor):
    def __init__(self):
        self.bindings = []
        self.class_globals = []
        self.class_depth = 0
        self.condition_depth = 0
        self.definitions = []
    def note(self, name, node, kind):
        self.bindings.append(dict(name=name, kind=kind, line_start=node.lineno, line_end=node.end_lineno, class_depth=self.class_depth))
    def visit_Name(self, node):
        if isinstance(node.ctx, (ast.Store, ast.Del)):
            self.note(node.id, node, type(node.ctx).__name__)
    def visit_Import(self, node):
        for a in node.names:
            self.note(a.asname or a.name.split('.')[0], node, 'import')
    def visit_ImportFrom(self, node):
        for a in node.names:
            self.note(a.asname or a.name, node, 'import_from')
    def visit_Global(self, node):
        if self.class_depth:
            self.class_globals.append(dict(names=node.names, line_start=node.lineno, line_end=node.end_lineno))
    def visit_FunctionDef(self, node):
        self.note(node.name, node, 'function_definition')
        self.definitions.append(dict(name=node.name, line_start=node.lineno, line_end=node.end_lineno, class_depth=self.class_depth, conditional=self.condition_depth > 0, type_parameters_present=bool(getattr(node, 'type_params', [])), signature=signature(node)))
        for expr in node.decorator_list + node.args.defaults + [x for x in node.args.kw_defaults if x is not None]:
            self.visit(expr)
        for arg in node.args.posonlyargs + node.args.args + node.args.kwonlyargs + ([node.args.vararg] if node.args.vararg else []) + ([node.args.kwarg] if node.args.kwarg else []):
            if arg.annotation:
                self.visit(arg.annotation)
        if node.returns:
            self.visit(node.returns)
    visit_AsyncFunctionDef = visit_FunctionDef
    def visit_Lambda(self, node):
        for expr in node.args.defaults + [x for x in node.args.kw_defaults if x is not None]:
            self.visit(expr)
    def visit_ClassDef(self, node):
        self.note(node.name, node, 'class_definition')
        for expr in node.bases + [k.value for k in node.keywords] + node.decorator_list:
            self.visit(expr)
        self.class_depth += 1
        for stmt in node.body:
            self.visit(stmt)
        self.class_depth -= 1
    def conditional(self, node):
        self.condition_depth += 1
        self.generic_visit(node)
        self.condition_depth -= 1
    visit_If = conditional
    visit_For = conditional
    visit_AsyncFor = conditional
    visit_While = conditional
    visit_Try = conditional
    visit_TryStar = conditional
    visit_With = conditional
    visit_AsyncWith = conditional
    visit_Match = conditional
    def visit_ExceptHandler(self, node):
        if node.name:
            self.note(node.name, node, 'except_binding')
        self.generic_visit(node)
    def visit_MatchAs(self, node):
        if node.name:
            self.note(node.name, node, 'match_binding')
        self.generic_visit(node)
    def visit_MatchStar(self, node):
        if node.name:
            self.note(node.name, node, 'match_binding')
    def visit_MatchMapping(self, node):
        if node.rest:
            self.note(node.rest, node, 'match_binding')
        self.generic_visit(node)

results = []
for path in sorted((PACKET / 'source').glob('*.raw')):
    raw = path.read_bytes()
    text = raw.decode('utf-8')
    tree = ast.parse(text, filename=str(path))
    inv = Inventory()
    inv.visit(tree)
    results.append(dict(source_path=str(path.relative_to(PACKET)), sha256=hashlib.sha256(raw).hexdigest(), line_count=len(text.splitlines()), definitions=inv.definitions, bindings=inv.bindings, class_globals=inv.class_globals))
(OUT / 'source_facts.json').write_text(json.dumps(dict(python=sys.version, modules=results), indent=2) + '\n')
print(json.dumps(dict(python=sys.version, modules=[dict(source_path=m['source_path'],sha256=m['sha256'],lines=m['line_count'],top_functions=[(d['name'],d['line_start']) for d in m['definitions'] if not d['class_depth']],class_globals=m['class_globals']) for m in results]), indent=2))
