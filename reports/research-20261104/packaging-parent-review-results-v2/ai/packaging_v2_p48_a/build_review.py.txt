import ast

import json
import pathlib
import sys

PACKET = pathlib.Path('reports/research-20261104/packaging-parents-v2/packets/packaging-derived-cf2cbe2aec28')
OUT = pathlib.Path('runs/packaging-v2-review-p48-a-20261104')
source_facts = json.loads((OUT / 'source_facts.json').read_text())
cases_bytes = (PACKET / 'cases.json').read_bytes()
cases = json.loads(cases_bytes)
assert len(cases) == 12
assert {c['source_path'] for c in cases} == set(source_facts['modules'])
# These individual judgments were made by reading complete sources and the rubric.
judgments = {
    'case-7f94633837e662610ec5': ('supported', 'Exactly one unconditional top-level parse_marker definition exists at lines 252-253; no ambiguity rule applies.'),
    'case-59b797fb4b76830426cc': ('supported', 'The independently extracted signature matches: source is positional_or_keyword, annotated str, without a default; return is MarkerList; no decorators; synchronous.'),

def normalize_extracted(s):
    return dict(parameters=s['parameters'], returns_ast=s['return_annotation_ast'], decorators_ast=s['decorators'], is_async=s['is_async'])

items = []
for case in cases:
    module = source_facts['modules'][case['source_path']]
    assert hashlib.sha256((PACKET / case['source_path']).read_bytes()).hexdigest() == module['sha256'] == case['source_sha256']
    definitions = [d for d in module['definitions'] if d['name'] == case['subject']['qualname']]
    verdict, reasoning = judgments[case['id']]
    facts = dict(subject_path=case['subject']['path'], subject_name=case['subject']['qualname'], definition_count=len(definitions), unconditional=bool(definitions), target_rebinding=len(definitions) > 1, type_parameters_present=any(d['type_parameters_present'] for d in definitions))
    if case['kind'] == 'signature_equals':
        facts['actual_signature'] = normalize_extracted(definitions[0]['signature']) if len(definitions) == 1 else None
        if verdict != 'uncertain':
            matches = facts['actual_signature'] == case['predicate']['expected_signature']
            assert matches == (verdict == 'supported')
    if len(definitions) > 1:
        facts['observed_definitions'] = [dict(line_start=d['line_start'], line_end=d['line_end'], signature=normalize_extracted(d['signature'])) for d in definitions]
    evidence = [dict(source_path=case['source_path'], available=True, line_start=1, line_end=module['lines'])]
    evidence += [dict(source_path=case['source_path'], available=True, line_start=d['line_start'], line_end=d['line_end']) for d in definitions]
    notes = f"Inspected the complete supplied module, lines 1-{module['lines']}, including imports, class bodies, and definition expressions. "
    if len(definitions) > 1:
        notes += 'Repeated definitions rebind the target; no other target bindings were found. The rubric forbids choosing the last definition.'
    else:
        notes += 'No conditional/duplicate target definitions, target rebinding, class-body target globals, wildcard imports, or subject type parameters found. Syntactic scope only; no runtime or behavioral claim.'
    items.append(dict(id=case['id'], source_sha256=case['source_sha256'], verdict=verdict, extracted_facts=facts, reasoning=reasoning, evidence=evidence, scope_notes=notes))

review = dict(schema_version=1, reviewer_kind='ai_reviewer', human=False, provider='codex', model='codex:gpt-6.1-sol', reasoning_effort='high', agent_task='/root/packaging_v2_p48_a', packet_sha256='77c4a11d4639a0590c9c22368b7683801568be064b3dcdb1ce101b36a654ce00', independence_disclosure='Model weights and training independence are unverified.', identity_basis='declared_invocation', items=items)
(OUT / 'review.json').write_text(json.dumps(review, indent=2) + '\n')
